```typescript
export const useMonitorStore = defineStore('monitor', () => {
  const events = ref<CommandEvent[]>([])
  const pendingApprovals = ref<ApprovalRequest[]>([])
  const socketConnected = ref(false)
  
  // 实时事件更新
  const addEvent = (event: CommandEvent) => {
```

```typescript
// 命令处理核心流程
@Injectable()
export class CommandService {
  async evaluateCommand(
    command: string,
    agentType: string,
    sessionId: string
  ): Promise<EvaluationResult> {
    // 1. AST解析
    const ast = this.parseCommand(command)
    
    // 2. 规则评估
    const ruleResults = await this.ruleService.evaluate(ast)
    
    // 3. 风险等级计算
    const riskLevel = this.calculateRisk(ruleResults)
    
    // 4. 决策逻辑
    if (riskLevel === 'block') return { action: 'deny', reason: 'Blocked by rules' }
    if (riskLevel === 'review') return await this.createApproval(command, ast)
    if (riskLevel === 'warn') return { action: 'allow', reason: 'Warning logged' }
    
    return { action: 'allow', reason: 'Safe command' }
  }
}
```

## Hook集成架构

### Claude Code Hook集成
```javascript
// ~/.claude/settings.json
{
  "hooks": {
    "preToolUse": "/path/to/aegis/hooks/universal-hook-v2.js"
  }
}

// universal-hook-v2.js 核心逻辑
async function preToolUse(toolCall) {
  if (toolCall.type !== 'bash') return null
  
  try {
    const response = await fetch('http://localhost:3000/api/command/evaluate', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        command: toolCall.parameters.command,
        agentType: 'claude-code',
        sessionId: getSessionId()
      })
    })
    
    const result = await response.json()
    
    if (result.action === 'deny') {
      return { block: true, reason: result.reason }
    }
    
    return null // 允许执行
  } catch (error) {
    console.error('[Aegis] Hook failed:', error)
    return null // 失败时允许执行，避免阻塞AI
  }
}
```

### 规则评估引擎
```typescript
@Injectable()
export class RuleEngine {
  private rules: Rule[] = []
  
  // 加载所有规则文件
  async loadRules(): Promise<void> {
    const ruleFiles = await this.scanRuleFiles()
    this.rules = []
    
    for (const file of ruleFiles) {
      const content = await fs.readFile(file, 'utf-8')
      const config = yaml.parse(content)
      this.rules.push(...config.rules)
    }
  }
  
  // 评估单个命令
  async evaluate(command: string): Promise<RuleEvaluation> {
    const ast = this.parseCommand(command)
    const results: RuleMatch[] = []
    
    for (const rule of this.rules) {
      const match = this.matchRule(rule, ast)
      if (match.matched) {
        results.push({
          ruleId: rule.id,
          severity: rule.severity,
          action: rule.action,
          reason: rule.reason,
          confidence: match.confidence
        })
      }
    }
    
    return this.calculateFinalAction(results)
  }
  
  // 命令AST解析
  private parseCommand(command: string): CommandAST {
    const parsed = shellQuote.parse(command)
    
    return {
      binary: parsed[0],
      subcommands: this.extractSubcommands(parsed),
      flags: this.extractFlags(parsed),
      arguments: this.extractArguments(parsed),
      fullCommand: command
    }
  }
}
```

```typescript
  for (const event of events) {

      event.id, event.command, event.agentType, 
      event.sessionId, event.status, event.riskLevel,
      event.reason, Date.now()
    )
  }
})

// 规则缓存优化
@Injectable()
export class RuleCache {
  private cache = new Map<string, RuleEvaluation>()
  private readonly TTL = 5 * 60 * 1000 // 5分钟
  
  get(commandHash: string): RuleEvaluation | null {
    const cached = this.cache.get(commandHash)
    if (cached && Date.now() - cached.timestamp < this.TTL) {
      return cached
    }
    this.cache.delete(commandHash)
    return null
  }
  
  set(commandHash: string, result: RuleEvaluation): void {
    this.cache.set(commandHash, {
      ...result,
      timestamp: Date.now()
    })
  }
}
```

```typescript
// 服务启动流程
export class AegisApp {
  async start(options: StartOptions = {}) {
    // 1. 检查端口可用性
    await this.checkPorts([3000, 3001])
    
    // 2. 初始化SQLite数据库
    await this.initDatabase()
    
    // 3. 启动NestJS后端 (端口3000)
    const app = await NestFactory.create(AppModule)
    app.enableCors()
    app.useWebSocketAdapter(new IoAdapter(app))
    await app.listen(3000)
    
    // 4. 启动Vue前端 (端口3001)
    // 静态文件已编译到 frontend/dist/
    app.useStaticAssets(path.join(__dirname, '../frontend/dist'))
    
    // 5. 输出启动信息
    console.log('🛡️  Aegis is running!')
    console.log('📊 Dashboard: http://localhost:3001')
    console.log('🔌 API: http://localhost:3000')
    
    // 6. 自动打开浏览器 (可选)
    if (!options.headless) {
      const open = await import('open')
      await open.default('http://localhost:3001')
    }
  }
}
```
