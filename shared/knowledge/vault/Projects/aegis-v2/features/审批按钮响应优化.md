## 问题背景
Aegis AI安全监控系统中，当AI Agent执行危险命令需要人工审批时，监控仪表板的批准/拒绝按钮点击后无响应。这会导致AI Agent无限等待，严重影响开发效率。主要问题：
1. WebSocket连接不稳定，审批消息无法正确传递
2. 前端审批状态管理混乱
3. Hook响应超时机制缺失

## 解决方案

      {{ isProcessing ? '处理中...' : '批准' }}

      :disabled="isProcessing"

<script setup lang="ts">
import { ref } from 'vue'
import { useApprovalStore } from '@/stores/approval'
import { showToast } from '@/utils/toast'

const approvalStore = useApprovalStore()
const isProcessing = ref(false)

const handleApprove = async () => {
  try {
    isProcessing.value = true
    await approvalStore.approveRequest(props.requestId)
    showToast('审批成功', 'success')
    emit('approved')
  } catch (error) {
    showToast('审批失败: ' + error.message, 'error')
  } finally {
    isProcessing.value = false
  }
}

}
</script>

### API 优化
```typescript
// 前端 Store
export const useApprovalStore = defineStore('approval', () => {
  const approveRequest = async (requestId: string) => {
    const response = await api.post(`/approval/${requestId}/approve`, {
      timeout: 10000, // 10秒超时
      retry: 2 // 重试2次
    })
    
    if (!response.ok) {
      throw new Error(response.data.message || '审批失败')
    }
    
    return response.data
  }
  
  return { approveRequest }
})

// 后端优化
app.post('/approval/:id/approve', async (req, res) => {
  const { id } = req.params
  const { user } = req.auth
  
  try {
    // 添加防重复提交检查
    const existing = await ApprovalRequest.findById(id)
    if (existing.status !== 'pending') {
      return res.status(409).json({ 
        error: '请求已处理，请刷新页面' 
      })
    }
    
    // 执行审批逻辑
    const result = await approvalService.processApproval(id, user.id, 'approved')
    
    // 实时通知
    await notificationService.notify({
      type: 'approval_processed',
      requestId: id,
      status: 'approved'
    })
    
    res.json({ success: true, data: result })
  } catch (error) {
    logger.error('Approval failed:', error)
    res.status(500).json({ 
      error: '系统错误，请稍后重试' 
    })
  }
})
```
